2. GERAR SAFE ZONES

Novo módulo:

safeZoneEngine.js
export function generateSafeZones(imageWidth, imageHeight, faces) {
  const zones = [];

  const margin = 40;

  const forbiddenZones = faces.map(face => ({
    x1: face.x - margin,
    y1: face.y - margin,
    x2: face.x + face.width + margin,
    y2: face.y + face.height + margin
  }));

  // zonas padrão candidatas
  const candidates = [
    { name: "top", x: 0, y: 0, w: imageWidth, h: imageHeight * 0.25 },
    { name: "bottom", x: 0, y: imageHeight * 0.75, w: imageWidth, h: imageHeight * 0.25 },
    { name: "left", x: 0, y: 0, w: imageWidth * 0.35, h: imageHeight },
    { name: "right", x: imageWidth * 0.65, y: 0, w: imageWidth * 0.35, h: imageHeight }
  ];

  function intersects(a, b) {
    return !(a.x > b.x2 || a.x + a.w < b.x1 || a.y > b.y2 || a.y + a.h < b.y1);
  }

  for (const zone of candidates) {
    const isSafe = !forbiddenZones.some(fz => intersects(zone, fz));
    if (isSafe) zones.push(zone);
  }

  return zones;
}

import { generateSafeZones } from "./safeZoneEngine.js";
import { detectFaces } from "../services/visionService.js";

export async function resolveLayout(imagePath, imageMeta) {
  const faces = await detectFaces(imagePath);

  const safeZones = generateSafeZones(
    imageMeta.width,
    imageMeta.height,
    faces
  );

  // prioridade estratégica
  const priority = ["bottom", "top", "right", "left"];

  const selectedZone = priority.find(p =>
    safeZones.find(z => z.name === p)
  );

  return selectedZone || { name: "bottom" }; // fallback
}

attentionEngine.js
export function simulateAttentionMap(scene) {
  const map = [];

  // prioridade 1: rosto
  if (scene.faces?.length) {
    scene.faces.forEach(face => {
      map.push({
        type: "face",
        weight: 1.0,
        area: expand(face, 60)
      });
    });
  }

  // prioridade 2: sujeito principal
  if (scene.subjectPosition) {
    map.push({
      type: "subject",
      weight: 0.8,
      area: scene.subjectPosition
    });
  }

  // prioridade 3: áreas de luz (simulação)
  if (scene.lighting === "high_contrast") {
    map.push({
      type: "light_hotspot",
      weight: 0.6,
      area: {
        x: scene.width * 0.6,
        y: scene.height * 0.2,
        w: 200,
        h: 200
      }
    });
  }

  return map;
}

function expand(area, padding) {
  return {
    x: area.x - padding,
    y: area.y - padding,
    w: area.w + padding * 2,
    h: area.h + padding * 2
  };
}
🧠 2. SAFE ZONE V3 (AGORA COM PESO)
safeZoneEngine.js (upgrade)
export function generateSafeZonesV3(width, height, attentionMap) {
  const zones = [
    { name: "top", x: 0, y: 0, w: width, h: height * 0.25 },
    { name: "bottom", x: 0, y: height * 0.75, w: width, h: height * 0.25 },
    { name: "left", x: 0, y: 0, w: width * 0.35, h: height },
    { name: "right", x: width * 0.65, y: 0, w: width * 0.35, h: height }
  ];

  return zones.map(zone => {
    let risk = 0;

    for (const attention of attentionMap) {
      if (intersects(zone, attention.area)) {
        risk += attention.weight;
      }
    }

    return {
      ...zone,
      risk
    };
  }).sort((a, b) => a.risk - b.risk);
}

function intersects(a, b) {
  return !(a.x > b.x + b.w || a.x + a.w < b.x || a.y > b.y + b.h || a.y + a.h < b.y);
}

export function renderText({ ctx, text, zone, strategy }) {
  if (strategy === "background_overlay") {
    drawBlurBox(ctx, zone);
  }

  drawText(ctx, text, zone);
}

function drawBlurBox(ctx, zone) {
  ctx.fillStyle = "rgba(0,0,0,0.5)";
  ctx.fillRect(zone.x, zone.y, zone.w, zone.h);
}
